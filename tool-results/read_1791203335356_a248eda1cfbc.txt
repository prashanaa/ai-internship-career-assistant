  1049→    function checkSkills() {
  1050→
  1051→        var role =
  1052→            document.getElementById("role").value;
  1053→
  1054→        var skills =
  1055→            document.getElementById("userSkills")
  1056→                    .value
  1057→                    .toLowerCase();
  1058→
  1059→
  1060→        var required = [];
  1061→
  1062→
  1063→        if (role == "data") {
  1064→
  1065→            required = [
  1066→                "excel",
  1067→                "sql",
  1068→                "power bi",
  1069→                "python"
  1070→            ];
  1071→
  1072→        }
  1073→
  1074→        else if (role == "web") {
  1075→
  1076→            required = [
  1077→                "html",
  1078→                "css",
  1079→                "javascript",
  1080→                "php"
  1081→            ];
  1082→
  1083→        }
  1084→
  1085→        else if (role == "hr") {
  1086→
  1087→            required = [
  1088→                "communication",
  1089→                "excel",
  1090→                "recruitment",
  1091→                "ms office"
  1092→            ];
  1093→
  1094→        }
  1095→
  1096→        else if (role == "marketing") {
  1097→
  1098→            required = [
  1099→                "seo",
  1100→                "social media",
  1101→                "content"
  1102→            ];
  1103→
  1104→        }
  1105→
  1106→
  1107→        var matched = 0;
  1108→
  1109→        var missing = [];
  1110→
  1111→
  1112→        for (var i = 0;
  1113→             i < required.length;
  1114→             i++) {
  1115→
  1116→            if (skills.includes(required[i])) {
  1117→
  1118→                matched++;
  1119→
  1120→            } else {
  1121→
  1122→                missing.push(required[i]);
  1123→
  1124→            }
  1125→
  1126→        }
  1127→
  1128→
  1129→        var percentage =
  1130→            Math.round(
  1131→                (matched / required.length) * 100
  1132→            );
  1133→
  1134→
  1135→        document.getElementById("skillResult")
  1136→                .innerHTML =
  1137→
  1138→                "<h3>Skill Match: " +

  1142→                "<p><b>Required Skills:</b> " +
  1143→                required.join(", ") +
  1144→                "</p>" +

  1148→                missing.join(", ") :

  1152→    }
